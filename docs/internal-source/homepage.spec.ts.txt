import { test as base, expect, type Page } from "@playwright/test";
import { mkdir } from "node:fs/promises";
import facts from "../data/business-facts.json";
import { services, gallery } from "../data/site";

// Every test navigates to the real Next.js server; no HTML parity preview is used.
const test = base.extend<{ diagnostics: void }>({
  diagnostics: [async ({ page }, use) => {
    const errors: string[] = [];
    page.on("pageerror", e => errors.push(`pageerror: ${e.message}`));
    page.on("console", msg => { if (msg.type() === "error") errors.push(`console: ${msg.text()}`); });
    page.on("response", r => { if (r.status() >= 400) errors.push(`HTTP ${r.status()}: ${r.url()}`); });
    page.on("requestfailed", r => {
      if (r.failure()?.errorText !== "net::ERR_ABORTED") errors.push(`request: ${r.url()} ${r.failure()?.errorText}`);
    });
    await use();
    expect(errors, "No JavaScript, hydration, console, or HTTP failures").toEqual([]);
  }, { auto: true }],
});

async function ready(page: Page) {
  const response = await page.goto("/");
  expect(response?.status()).toBe(200);
  await page.evaluate(() => document.fonts.ready);
  await expect(page.locator("h1")).toBeVisible();
}

async function noOverflow(page: Page) {
  expect(await page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth)).toBe(0);
  const clippedControls = await page.locator(".page-content a:visible, .page-content button:visible").evaluateAll(items =>
    items.filter(item => { const r = item.getBoundingClientRect(); return r.left < -1 || r.right > innerWidth + 1; })
      .map(item => item.textContent?.trim()));
  expect(clippedControls, "No actionable controls clipped by the viewport").toEqual([]);
}

test("real Next.js render, factual content, assets, RTL and review screenshots", async ({ page }, info) => {
  await ready(page);
  await expect(page.locator("html")).toHaveAttribute("lang", "fa");
  await expect(page.locator("html")).toHaveAttribute("dir", "rtl");
  await expect(page.locator('meta[name="robots"]')).toHaveAttribute("content", /noindex/);
  expect(await page.title()).toBe("محمد استایل | سالن ممتاز پونک");
  expect(await page.locator(".service-card strong").allTextContents()).toEqual(facts.verified_or_user_provided.services);
  await expect(page.locator(".service-concept-label")).toHaveCount(4);
  for (const id of ["facial", "massage", "manicure", "pedicure"]) {
    await expect(page.locator(`.service-${id} img`)).toHaveAttribute("alt", /تصویر مفهومی/);
    await expect(page.locator(`.service-${id} .service-concept-label`)).toHaveText("تصویر مفهومی");
  }
  await expect(page.locator(".branch-known")).toContainText(facts.verified_or_user_provided.branch_1_location);
  await expect(page.locator(".branch-pending button, .branch-pending a, .branch-pending img")).toHaveCount(0);

  const links = await page.locator("a[href]").evaluateAll(items => items.map(item => item.getAttribute("href")!));
  for (const href of links) {
    if (href.startsWith("#")) await expect(page.locator(href)).toHaveCount(1);
    else expect(href).toBe("https://www.instagram.com/mohammadstyle1/");
  }
  // Scroll every lazy image into view, then await its real browser decode.
  for (const img of await page.locator("img").all()) {
    await img.scrollIntoViewIfNeeded();
    await img.evaluate((node: HTMLImageElement) => node.decode());
  }
  expect(await page.locator("img").evaluateAll(items => items.map(item => item as HTMLImageElement).filter(img => !img.complete || !img.naturalWidth).map(img => img.src))).toEqual([]);
  const fonts = await page.evaluate(() => ({
    heading: getComputedStyle(document.querySelector("h1")!).fontFamily,
    body: getComputedStyle(document.body).fontFamily,
    loaded: [...document.fonts].filter(font => /^(kahroba|iransans)$/.test(font.family)).every(font => font.status === "loaded"),
  }));
  expect(fonts.heading.toLowerCase()).toContain("kahroba");
  expect(fonts.body.toLowerCase()).toContain("iransans");
  expect(fonts.loaded).toBe(true);
  await page.evaluate(() => scrollTo(0, 0));
  await noOverflow(page);
  await mkdir("artifacts/screenshots/codex", { recursive: true });
  const prefix = `artifacts/screenshots/codex/homepage-${info.project.name}`;
  await page.screenshot({ path: `${prefix}-above-fold.png` });
  await page.screenshot({ path: `${prefix}-full.png`, fullPage: true });
  await info.attach("render-diagnostics", { body: JSON.stringify({ url: page.url(), viewport: page.viewportSize(), fonts, imageCount: await page.locator("img").count(), horizontalOverflow: 0 }), contentType: "application/json" });
});

test("portfolio filters and Before/After pointer and keyboard operation", async ({ page }, info) => {
  await ready(page);
  const overviewCraftWidth = (await page.locator(".gallery-item-craft").boundingBox())!.width;
  const labels = ["همه", "قبل و بعد", "استایل دامادی", "پشت صحنه"];
  const categories = ["all", "beforeafter", "groom", "behind"];
  for (let i = 0; i < labels.length; i++) {
    const button = page.locator(".gallery-toolbar").getByRole("button", { name: labels[i], exact: true });
    await button.click();
    await expect(button).toHaveAttribute("aria-pressed", "true");
    const expected = categories[i] === "all" ? gallery : gallery.filter(g => g.category === categories[i]);
    expect(await page.locator(".gallery-item strong").allTextContents()).toEqual(expected.map(g => g.title));
    if (categories[i] === "groom") {
      await expect(page.locator('.gallery-item img[src="/images/groom-look.webp"]')).toHaveCount(0);
    }
    if (categories[i] === "behind") {
      await expect(page.locator(".gallery-item-barber")).toContainText("آرایشگر مجموعه");
      if (info.project.name === "desktop") {
        const filteredCraftWidth = (await page.locator(".gallery-item-craft").boundingBox())!.width;
        expect(filteredCraftWidth / overviewCraftWidth, "Desktop behind-the-scenes must not magnify the craft photo").toBeLessThanOrEqual(1.01);
      }
    }
    await expect(page.getByRole("status")).toContainText(`${expected.length} نمونه‌کار`);
    for (const img of await page.locator(".gallery-item img").all()) await img.evaluate((node: HTMLImageElement) => node.decode());
    await noOverflow(page);
  }
  await page.locator(".gallery-toolbar").getByRole("button", { name: "همه", exact: true }).click();
  const slider = page.getByRole("slider");
  await slider.scrollIntoViewIfNeeded();
  await slider.focus();
  await slider.press("End");
  await expect(slider).toHaveValue("100");
  await expect(page.locator(".compare-frame")).toHaveAttribute("style", /100%/);
  await expect(page.locator(".compare-images")).toHaveCSS("outline-style", "solid");
  await slider.press("Home");
  await expect(slider).toHaveValue("0");
  await slider.press("ArrowRight");
  await expect(slider).toHaveValue("1");
  await expect(slider).toHaveAttribute("aria-valuetext", "1٪ قبل، 99٪ بعد");
  const bounds = (await slider.boundingBox())!;
  if (info.project.name.startsWith("mobile")) await page.touchscreen.tap(bounds.x + bounds.width * .7, bounds.y + bounds.height / 2);
  else {
    await page.mouse.move(bounds.x + bounds.width * .5, bounds.y + bounds.height / 2);
    await page.mouse.down();
    await page.mouse.move(bounds.x + bounds.width * .7, bounds.y + bounds.height / 2);
    await page.mouse.up();
  }
  expect(Number(await slider.inputValue())).toBeGreaterThan(60);
  expect(Number(await slider.inputValue())).toBeLessThan(80);
});

test("service preview, drawer focus loop, isolation and all close methods", async ({ page }, info) => {
  await ready(page);
  const dialog = page.getByRole("dialog");
  const close = dialog.getByRole("button", { name: "بستن", exact: true });
  for (const service of services) {
    const card = page.locator(`.service-${service.id}`);
    await card.click();
    await expect(dialog).toBeVisible();
    await expect(close).toBeFocused();
    await expect(page.locator(".page-content")).toHaveAttribute("inert", "");
    await expect(page.locator("body")).toHaveCSS("overflow", "hidden");
    await expect(dialog.getByRole("button", { name: service.name, exact: true })).toHaveAttribute("aria-pressed", "true");
    await expect(dialog.getByRole("button", { name: "ادامه رزرو (غیرفعال)" })).toBeDisabled();
    expect(await dialog.locator('input[type="date"], input[type="time"], form').count()).toBe(0);
    await expect(dialog.locator(".drawer-info")).toContainText("هیچ نوبتی ثبت نمی‌شود");
    await page.keyboard.press("Escape");
    await expect(dialog).toHaveCount(0);
    await expect(card).toBeFocused();
    await expect(page.locator(".page-content")).not.toHaveAttribute("inert");
    await expect(page.locator("body")).not.toHaveCSS("overflow", "hidden");
  }
  const opener = page.locator(".header-book");
  await opener.click();
  await expect(dialog.locator('[aria-pressed="true"]')).toHaveCount(0);
  await expect(close).toBeFocused();
  await page.keyboard.press("Shift+Tab");
  await expect(dialog.getByRole("button", { name: "پدیکور", exact: true })).toBeFocused();
  await page.keyboard.press("Tab");
  await expect(close).toBeFocused();
  await dialog.getByRole("button", { name: "هیرکات مو", exact: true }).click();
  await expect(dialog.getByRole("button", { name: "هیرکات مو", exact: true })).toHaveAttribute("aria-pressed", "true");
  await close.focus();
  await page.screenshot({ path: `artifacts/screenshots/codex/drawer-${info.project.name}.png` });
  await close.click();
  await expect(opener).toBeFocused();
  if (!info.project.name.startsWith("mobile")) {
    await opener.click();
    await page.locator(".drawer-scrim").click({ position: { x: 20, y: 50 } });
    await expect(dialog).toHaveCount(0);
    await expect(opener).toBeFocused();
  }
  await page.locator(".branch-known button").click();
  await expect(dialog).toContainText(facts.verified_or_user_provided.branch_1_location);
  await page.keyboard.press("Escape");
  await page.locator(".closing-section button").click();
  await expect(dialog).toBeVisible();
  await page.keyboard.press("Escape");
  await noOverflow(page);
});

test("navigation, mobile menu Escape and focus, skip link", async ({ page }, info) => {
  await ready(page);
  await page.keyboard.press("Tab");
  await expect(page.locator(".skip-link")).toBeFocused();
  await page.keyboard.press("Enter");
  await expect(page.locator("main")).toBeFocused();
  if (info.project.name.startsWith("mobile")) {
    const menu = page.locator(".menu-button");
    await menu.click();
    await expect(page.locator("#primary-nav")).toBeVisible();
    await expect(menu).toHaveAttribute("aria-expanded", "true");
    await page.screenshot({ path: `artifacts/screenshots/codex/menu-${info.project.name}.png` });
    await page.keyboard.press("Escape");
    await expect(menu).toBeFocused();
    await expect(menu).toHaveAttribute("aria-expanded", "false");
    await expect(page.locator("#primary-nav")).toBeHidden();
    for (const target of ["#home", "#services", "#portfolio", "#branches", "#about"]) {
      await menu.click();
      await page.locator(`#primary-nav a[href="${target}"]`).click();
      await expect(page.locator("#primary-nav")).toBeHidden();
      await expect(page).toHaveURL(new RegExp(`${target}$`));
      await noOverflow(page);
    }
    await menu.click();
    await page.locator(".hero-ctas button").click();
    await page.keyboard.press("Escape");
    await expect(page.locator("#primary-nav")).toBeHidden();
  } else {
    for (const target of ["#home", "#services", "#portfolio", "#branches", "#about"]) {
      await page.locator(`#primary-nav a[href="${target}"]`).click();
      await expect(page).toHaveURL(new RegExp(`${target}$`));
    }
  }
  await page.locator(".hero-ctas a").click();
  await expect(page).toHaveURL(/#portfolio$/);
  await noOverflow(page);
});
